cmake_minimum_required(VERSION 3.22)
project(PlasmaNextDock LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
include(GNUInstallDirs)
include(CTest)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Quick Qml WaylandClient)
# Recent Qt releases no longer import private targets through public modules.
# Older releases provide GuiPrivate through Gui without a separate package.
if(NOT TARGET Qt6::GuiPrivate)
    find_package(Qt6 REQUIRED COMPONENTS GuiPrivate)
endif()
find_package(LayerShellQt REQUIRED)
find_package(PlasmaWaylandProtocols REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(WaylandClient REQUIRED IMPORTED_TARGET wayland-client)
add_library(dockcore STATIC src/geometry.cpp src/controller.cpp src/settings.cpp src/popupanchor.cpp)
target_include_directories(dockcore PUBLIC src)
target_link_libraries(dockcore PUBLIC Qt6::Quick)
add_executable(plasma-next-dock src/main.cpp src/screenedge.cpp src/dockhost.cpp)
target_link_libraries(plasma-next-dock PRIVATE dockcore Qt6::WaylandClient Qt6::GuiPrivate LayerShellQt::Interface PkgConfig::WaylandClient)
set(wayland_codegen_options)
if(Qt6_VERSION VERSION_GREATER_EQUAL 6.12)
    # Preserve the public generator API used by the adapter, explicitly (QTP0006).
    list(APPEND wayland_codegen_options PUBLIC_CODE)
endif()
qt6_generate_wayland_protocol_client_sources(plasma-next-dock ${wayland_codegen_options} FILES
    ${PLASMA_WAYLAND_PROTOCOLS_DIR}/kde-screen-edge-v1.xml)
set(DOCK_QML_FILES
    qml/Settings.qml qml/Main.qml qml/LiveTasks.qml qml/DemoTasks.qml qml/CapsuleGlow.qml qml/code/CapsuleGeometry.js qml/code/InteractionGeometry.js qml/code/AdaptiveLayout.js)
qt_add_resources(plasma-next-dock dockqml PREFIX / FILES ${DOCK_QML_FILES})
install(TARGETS plasma-next-dock RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
# Generate Exec at install time so --prefix overrides work too. The portal's
# PATH need not contain a user's ~/.local/bin.
configure_file(cmake/InstallDesktop.cmake.in InstallDesktop.cmake @ONLY)
install(SCRIPT "${CMAKE_CURRENT_BINARY_DIR}/InstallDesktop.cmake")
if(BUILD_TESTING)
    find_package(Qt6 REQUIRED COMPONENTS Test QuickTest)
    add_executable(hoverqmltest tests/hoverqmltest.cpp)
    target_link_libraries(hoverqmltest PRIVATE dockcore Qt6::QuickTest)
    qt_add_resources(hoverqmltest dockqml PREFIX / FILES ${DOCK_QML_FILES})
    add_test(NAME hover-and-render COMMAND hoverqmltest -input ${CMAKE_CURRENT_SOURCE_DIR}/tests/qml)
    set_tests_properties(hover-and-render PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    add_executable(settingstest tests/settingstest.cpp)
    target_link_libraries(settingstest PRIVATE dockcore Qt6::Test)
    add_test(NAME settings-persistence COMMAND settingstest)
    add_test(NAME settings-window-smoke COMMAND plasma-next-dock --demo --smoke-test --settings)
    set_tests_properties(settings-window-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    add_executable(coretest tests/coretest.cpp)
    target_link_libraries(coretest PRIVATE dockcore Qt6::Test)
    add_test(NAME geometry-and-dodge COMMAND coretest)
    set_tests_properties(geometry-and-dodge PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
    foreach(edge bottom top left right)
        add_test(NAME qml-${edge} COMMAND plasma-next-dock --demo --smoke-test --glow --edge ${edge})
        set_tests_properties(qml-${edge} PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
    endforeach()
    add_test(NAME live-model-smoke COMMAND plasma-next-dock --smoke-test)
    set_tests_properties(live-model-smoke PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software")
endif()
